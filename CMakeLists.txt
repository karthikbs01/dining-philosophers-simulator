cmake_minimum_required(VERSION 3.16)
project(DiningPhilosophers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(ENABLE_TSAN "Build with ThreadSanitizer on supported Unix compilers" OFF)
if(ENABLE_TSAN)
    if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Use a Linux GCC/Clang toolchain for ThreadSanitizer")
    endif()
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=thread)
endif()
add_library(engine simulation.cpp resource_manager.cpp philosopher.cpp)
target_include_directories(engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(engine PUBLIC Threads::Threads)
add_executable(dining main.cpp)
target_link_libraries(dining PRIVATE engine)
include(CTest)
if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE engine)
    add_test(NAME synchronization COMMAND engine_tests)
    set_tests_properties(synchronization PROPERTIES TIMEOUT 90)
endif()

option(BUILD_WEBSITE "Build the local HTTP/SSE server" ON)
if(BUILD_WEBSITE)
    add_executable(dining_server website/backend/server.cpp)
    target_link_libraries(dining_server PRIVATE engine)
    if(WIN32)
        target_link_libraries(dining_server PRIVATE ws2_32)
    endif()
    if(MINGW)
        target_compile_options(dining_server PRIVATE -Wa,-mbig-obj)
    endif()
endif()
if(MINGW)
    # Keep the Windows executables runnable from an ordinary PowerShell session.
    target_link_options(engine INTERFACE -static)
endif()
